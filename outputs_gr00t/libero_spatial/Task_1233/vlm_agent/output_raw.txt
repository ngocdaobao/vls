# Task breakdown:
# Stage 1: Move gripper to grasp the dark vessel (akita_black_bowl_1_main) near the biscuit container (keypoint 0)
# Stage 2: Move gripper with the dark vessel to set it on the dish (plate_1_main) at keypoints 6 and 7

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # From the image, the dark vessel (akita_black_bowl_1_main) is at keypoint 0
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # From the image, the dish (plate_1_main) is at keypoints 6 and 7
    # We'll use the mean of the two keypoints as the target location
    plate_idx = torch.tensor([6, 7], dtype=torch.long, device=keypoints.device)
    plate_pos = keypoints[plate_idx].mean(dim=0)
    all_dists = torch.norm(action_sequence - plate_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

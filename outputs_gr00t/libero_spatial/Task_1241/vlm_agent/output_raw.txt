# Task breakdown:
# Stage 1: Move gripper to grasp the black bowl that is next to the cookie box (keypoint 3)
# Stage 2: Move gripper with the black bowl to the plate (keypoints 10, 11, 12) and place it gently

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # The black bowl next to the cookie box is marked by keypoint 3
    target_idx = torch.tensor([3], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # The plate is marked by keypoints 10, 11, 12 - use their mean as target
    plate_indices = torch.tensor([10, 11, 12], dtype=torch.long, device=keypoints.device)
    plate_pos = keypoints[plate_indices].mean(dim=0)
    all_dists = torch.norm(action_sequence - plate_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

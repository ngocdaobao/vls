# Task breakdown:
# Stage 1: Move gripper to grasp the darkhued dish (akita_black_bowl_1_main) positioned adjacent to the container of baked treats (keypoint 0)
# Stage 2: Move gripper with the grasped dish to set it atop the dish designed for individual servings (plate_1_main, keypoint 6)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # The darkhued dish (akita_black_bowl_1_main) is at keypoint 0, adjacent to the container of baked treats (keypoint 3)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # The dish designed for individual servings (plate_1_main) is at keypoint 6
    target_idx = torch.tensor([6], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

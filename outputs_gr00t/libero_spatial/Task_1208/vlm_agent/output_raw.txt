# Task breakdown:
# Stage 1: Move gripper to grasp the black bowl (keypoint 0) which is on the table
# Stage 2: Move gripper with the black bowl to place it next to the cookie box (keypoint 3) onto the plate (keypoints 6,7,8)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target is the location next to the cookie box on the plate
    # Cookie box is at keypoint 3, plate is at keypoints 6,7,8
    # We want to place the bowl next to the cookie box on the plate
    # Use the center of the plate as reference
    plate_center_idx = torch.tensor([6, 7, 8], dtype=torch.long, device=keypoints.device)
    plate_center = keypoints[plate_center_idx].mean(dim=0)
    cookie_box_pos = keypoints[torch.tensor([3], dtype=torch.long, device=keypoints.device)][0]
    # Define target as a point on the plate, next to the cookie box
    # We can use the plate center as the main target
    target_pos = plate_center

    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

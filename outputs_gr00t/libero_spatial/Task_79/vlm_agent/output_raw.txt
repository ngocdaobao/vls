# Task breakdown:
# Stage 1: Move gripper to grasp the akita black bowl on the top layer of the wooden cabinet (keypoint 1)
# Stage 2: Move gripper with the grasped bowl to place it on the plate (keypoints 8, 9)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Keypoint 1 is the akita black bowl on the top layer of the wooden cabinet
    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Plate is represented by keypoints 8 and 9; use their mean as target
    plate_idx = torch.tensor([8, 9], dtype=torch.long, device=keypoints.device)
    plate_pos = keypoints[plate_idx].mean(dim=0)
    all_dists = torch.norm(action_sequence - plate_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

# Task breakdown:
# Stage 1: Move gripper to grasp the black bowl adjacent to the cookie box (keypoint 0)
# Stage 2: Move gripper with the grasped black bowl to position it onto the plate (keypoints 6, 7)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Keypoint 0 is the black bowl adjacent to the cookie box (as seen in the image)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Keypoints 6 and 7 represent the plate (center of the plate is between them)
    plate_center_idx = torch.tensor([6, 7], dtype=torch.long, device=keypoints.device)
    plate_center_pos = keypoints[plate_center_idx].mean(dim=0)
    all_dists = torch.norm(action_sequence - plate_center_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

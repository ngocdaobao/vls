# Task breakdown:
# Stage 1: Move gripper to grasp the darkhued vessel (keypoint 3) which is adjacent to the container holding baked treats (keypoint 4)
# Stage 2: Move gripper with the darkhued vessel to the dish for serving food (keypoints 7, 8)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # The darkhued vessel is at keypoint 3, adjacent to the container holding baked treats (keypoint 4)
    target_idx = torch.tensor([3], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # The dish for serving food is the plate at keypoints 7 and 8
    # Use the center of the plate as target
    plate_center = (keypoints[torch.tensor([7, 8], dtype=torch.long, device=keypoints.device)]).mean(dim=0)
    target_pos = plate_center
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()
